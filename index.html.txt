    <script>
        function switchTab(tabId) {
            document.querySelectorAll('.studio-container').forEach(el => el.classList.remove('active-section'));
            document.querySelectorAll('nav button').forEach(el => el.classList.remove('active'));
            document.getElementById('section-' + tabId).classList.add('active-section');
            document.getElementById('nav-' + tabId).classList.add('active');
        }

        function generateImage() {
            const prompt = document.getElementById('prompt-art').value.trim();
            const style = document.getElementById('style-art').value;
            const container = document.getElementById('preview-art');
            if(!prompt) return alert('Introdu o descriere!');
            container.innerHTML = `<div class="preview-result"><div class="spinner"></div><p>Se generează imaginea...</p></div>`;
            const url = `https://image.pollinations.ai/prompt/${encodeURIComponent(prompt + ', ' + style)}?width=1024&height=768&nologo=true`;
            const img = new Image();
            img.src = url;
            img.onload = () => container.innerHTML = `<div class="preview-result"><img src="${url}"><a href="${url}" target="_blank" class="btn-primary" style="padding:0.5rem 1rem;font-size:0.85rem;">Descarcă</a></div>`;
        }

        let uploadedImageSrc = '';
        document.getElementById('image-upload').addEventListener('change', (e) => {
            const file = e.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = (event) => {
                    uploadedImageSrc = event.target.result;
                    document.getElementById('photo-preview').innerHTML = `<img src="${uploadedImageSrc}" id="target-img">`;
                };
                reader.readAsDataURL(file);
            }
        });

        function applyPhotoFilter() {
            if(!uploadedImageSrc) return alert('Încarcă o poză mai întâi!');
            document.getElementById('target-img').style.filter = document.getElementById('filter-select').value;
        }

        function generateVideo() {
            const p = document.getElementById('video-prompt').value;
            if(!p) return alert('Introdu un scenariu!');
            const cp = document.getElementById('video-preview');
            cp.innerHTML = `<div class="preview-result"><div class="spinner"></div><p>Se randarează video-ul...</p></div>`;
            setTimeout(() => cp.innerHTML = `<div class="preview-result"><p style="color:#818cf8;font-weight:bold;">▶️ Redare Video: "${p}"</p><p style="font-size:0.85rem;color:var(--text-muted);">[Generat cu succes]</p></div>`, 2000);
        }

        function generateMelodie() {
            const p = document.getElementById('melodie-prompt').value;
            const cp = document.getElementById('melodie-preview');
            cp.innerHTML = `<div class="preview-result"><div class="spinner"></div><p>Se compune piesa...</p></div>`;
            setTimeout(() => cp.innerHTML = `<div class="preview-result"><p style="color:#818cf8;font-weight:bold;">🎵 Piesă Generată: "${p}"</p><audio controls style="width:100%"><source src="https://www.soundhelix.com/examples/mp3/SoundHelix-Song-1.mp3" type="audio/mp3"></audio></div>`, 2000);
        }

        let mediaRec, chunks = [];
        async function toggleRecord() {
            const btn = document.getElementById('record-btn');
            const cp = document.getElementById('audio-preview');
            if (!mediaRec || mediaRec.state === "inactive") {
                try {
                    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
                    mediaRec = new MediaRecorder(stream);
                    chunks = [];
                    mediaRec.ondataavailable = e => chunks.push(e.data);
                    mediaRec.onstop = () => {
                        const blob = new Blob(chunks, { type: 'audio/mp3' });
                        cp.innerHTML = `<div class="preview-result"><p style="color:#818cf8;font-weight:bold;">Înregistrare finalizată!</p><audio controls src="${URL.createObjectURL(blob)}" style="width:100%"></audio></div>`;
                    };
                    mediaRec.start();
                    btn.textContent = "⏹️ Oprește";
                    btn.style.background = "#22c55e";
                } catch { alert('Eroare microfon.'); }
            } else {
                mediaRec.stop();
                btn.textContent = "🎙️ Înregistrează Vocea";
                btn.style.background = "#ef4444";
            }
        }
    </script>
